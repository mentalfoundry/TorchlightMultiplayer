# Party travel: the host takes the stairs down; the client is next to it, so after the countdown both
# arrive on the same new floor, run by the host, with the client's copies of its monsters.
players 2 timeout=90
click ContinueLast timeout=90
ingame timeout=90
sync ingame timeout=120
sync together timeout=60
mark
stairs down
expect notice Leaving in timeout=20
expect newlevel timeout=60
expect samelevel 1 timeout=60
expect owner 0 timeout=30
wait 6
sync arrived timeout=60
dump arrived
shot arrived
sync end timeout=60
quit
