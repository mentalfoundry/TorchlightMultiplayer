# Quests across games: both take DEFEAT quests through the console (the host's as a baseline), then the
# party goes down three floors; dumps record quest state and monsters on every floor.
players 2 timeout=90
click ContinueLast timeout=90
ingame timeout=90
sync ingame timeout=120
console QUESTACTIVE Bard_Quest1
console QUESTACTIVE BardRandom
wait 2
dump quests
expect quest Bard_Quest1 1 timeout=10
expect quest BardRandom 1 timeout=10
sync go1 timeout=60
mark
stairs down
expect newlevel timeout=60
expect samelevel 1 timeout=60
wait 6
sync f1 timeout=60
dump f1
expect quest Bard_Quest1 1 timeout=10
expect quest BardRandom 1 timeout=10
shot f1
sync go2 timeout=60
mark
stairs down
expect newlevel timeout=60
expect samelevel 1 timeout=60
wait 6
sync f2 timeout=60
dump f2
expect quest Bard_Quest1 1 timeout=10
expect quest BardRandom 1 timeout=10
shot f2
sync go3 timeout=60
mark
stairs down
expect newlevel timeout=60
expect samelevel 1 timeout=60
wait 6
sync f3 timeout=60
dump f3
expect quest Bard_Quest1 1 timeout=10
expect quest BardRandom 1 timeout=10
shot f3
sync end timeout=60
quit
