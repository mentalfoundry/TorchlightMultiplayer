# Combat across games: client hits count on the host, monsters hurt the client, kills give XP to both.
players 2 timeout=90
click ContinueLast timeout=90
ingame timeout=90
sync ingame timeout=120
wait 5
mark
sync start timeout=60
# The client fights first (near us: it walked to us).
sync clientkilled timeout=150
expect xpgain
sync dump1 timeout=30
dump afterclientkill
shot afterclientkill
# Now we kill one next to the client.
mark
teleport monster 0
sync hostready timeout=30
sync clientnear timeout=30
killnear timeout=30
expect xpgain
wait 3
sync dump2 timeout=60
dump afterhostkill
shot afterhostkill
sync end timeout=60
quit
