cmake_minimum_required(VERSION 3.24)

# Torchlight is a 32-bit process; this must be configured with -A Win32 and the
# x86-windows-static vcpkg triplet (see build.ps1).
project(TLMP LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Static CRT: the game ships the VS2008 CRT only, and we must not depend on a
# redistributable being installed on every player's machine.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "TLMP must be built as 32-bit (use -A Win32)")
endif()

find_package(minhook CONFIG REQUIRED)
find_package(unofficial-enet CONFIG REQUIRED)

add_library(TLMP SHARED
  src/plugin.cpp
  src/log.cpp
  src/config.cpp
  src/crashlog.cpp
  src/profile.cpp
  src/net/session.cpp
  src/game/game.cpp
  src/game/diag.cpp
  src/game/monsters.cpp
  src/game/combat.cpp
  src/game/anim.cpp
  src/game/loot.cpp
  src/game/floor.cpp
  src/game/savestate.cpp
  src/game/spawns.cpp
  src/game/autotest.cpp
  src/game/quests.cpp
  src/game/skills.cpp
  src/game/triggers.cpp
  src/game/party.cpp
  src/game/hooks.cpp
  src/TLMP.def
)
target_include_directories(TLMP PRIVATE src)
target_compile_definitions(TLMP PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(TLMP PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(TLMP PRIVATE minhook::minhook unofficial::enet::enet ws2_32 winmm)

# Forwarding proxy for the game's OgreGUIRenderer.dll; makes TLMP.dll load at process
# startup (see src/proxy/proxy.cpp). The .def is generated by tools/gen_proxy_def.py.
add_library(OgreGUIRenderer SHARED src/proxy/proxy.cpp src/proxy/OgreGUIRenderer.def)
target_compile_definitions(OgreGUIRenderer PRIVATE WIN32_LEAN_AND_MEAN)
target_link_libraries(OgreGUIRenderer PRIVATE TLMP)
